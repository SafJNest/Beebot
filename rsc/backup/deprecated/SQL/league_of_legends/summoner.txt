CREATE TABLE `summoner` (
 `id` int(11) NOT NULL AUTO_INCREMENT,
 `riot_id` varchar(255) DEFAULT NULL,
 `level` int(11) DEFAULT NULL,
 `icon` int(11) DEFAULT NULL,
 `region` varchar(255) DEFAULT NULL,
 `puuid` varchar(255) DEFAULT NULL,
 `user_id` varchar(255) DEFAULT NULL,
 `tracking` int(11) NOT NULL DEFAULT 0,
 `last_update` datetime NOT NULL DEFAULT current_timestamp(),
 `riot_search` varchar(255) GENERATED ALWAYS AS (lcase(replace(replace(`riot_id`,' ',''),'#',''))) STORED,
 PRIMARY KEY (`id`),
 UNIQUE KEY `unique_puuid_idx` (`puuid`,`region`) USING BTREE,
 KEY `user_idx` (`user_id`),
 KEY `tracking` (`tracking`),
 KEY `region` (`region`),
 KEY `riot_id_2` (`riot_id`),
 KEY `riot_search` (`riot_search`),
 FULLTEXT KEY `riot_id` (`riot_id`)
) ENGINE=InnoDB AUTO_INCREMENT=48309696 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
